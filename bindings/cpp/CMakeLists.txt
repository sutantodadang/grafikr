cmake_minimum_required(VERSION 3.28)
project(GrafikrCpp VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT GRAFIKR_FFI_LIBRARY)
  message(FATAL_ERROR "Set GRAFIKR_FFI_LIBRARY to the built Grafikr shared library")
endif()
include(GNUInstallDirs)
install(FILES include/grafikr.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
find_package(Threads REQUIRED)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../c ${CMAKE_BINARY_DIR}/grafikr-c)
add_library(grafikr_cpp INTERFACE)
target_include_directories(grafikr_cpp INTERFACE include ../c/include)
target_link_libraries(grafikr_cpp INTERFACE grafikr_client)
foreach(example counter_form dashboard multi_window)
  add_executable(grafikr_cpp_${example} examples/${example}.cpp)
  target_link_libraries(grafikr_cpp_${example} PRIVATE grafikr_cpp)
  if(NOT GRAFIKR_FFI_STATIC AND WIN32)
    add_custom_command(TARGET grafikr_cpp_${example} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GRAFIKR_FFI_LIBRARY}" "$<TARGET_FILE_DIR:grafikr_cpp_${example}>")
  endif()
endforeach()
install(TARGETS grafikr_cpp_counter_form grafikr_cpp_dashboard grafikr_cpp_multi_window
  RUNTIME DESTINATION ${CMAKE_INSTALL_DATADIR}/grafikr/examples)
include(CTest)
if(BUILD_TESTING)
  add_executable(grafikr_cpp_test tests/client.cpp)
  target_link_libraries(grafikr_cpp_test PRIVATE grafikr_cpp)
  if(NOT GRAFIKR_FFI_STATIC)
    add_custom_command(TARGET grafikr_cpp_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GRAFIKR_FFI_LIBRARY}" "$<TARGET_FILE_DIR:grafikr_cpp_test>")
  endif()
  add_test(NAME grafikr_cpp_test COMMAND grafikr_cpp_test)
  add_executable(grafikr_cpp_collection_provider_scenario tests/collection_provider_scenario.cpp)
  target_link_libraries(grafikr_cpp_collection_provider_scenario PRIVATE grafikr_cpp)
  add_executable(grafikr_cpp_transport_contract_test tests/transport_contract.cpp)
  target_include_directories(grafikr_cpp_transport_contract_test PRIVATE ../c/tests)
  target_compile_definitions(grafikr_cpp_transport_contract_test PRIVATE GRAFIKR_STATIC)
  target_link_libraries(grafikr_cpp_transport_contract_test PRIVATE grafikr_cpp grafikr_test_mock_ffi Threads::Threads)
  add_test(NAME grafikr_cpp_transport_contract_test COMMAND grafikr_cpp_transport_contract_test)
endif()
