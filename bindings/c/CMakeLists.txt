cmake_minimum_required(VERSION 3.28)
project(GrafikrC VERSION 1.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT GRAFIKR_FFI_LIBRARY)
  message(FATAL_ERROR "Set GRAFIKR_FFI_LIBRARY to the built Grafikr shared library")
endif()
option(GRAFIKR_FFI_STATIC "Link the Rust static library" OFF)
find_package(Threads REQUIRED)
if(GRAFIKR_FFI_STATIC)
  add_library(grafikr_ffi STATIC IMPORTED GLOBAL)
  set_target_properties(grafikr_ffi PROPERTIES IMPORTED_LOCATION "${GRAFIKR_FFI_LIBRARY}")
else()
  add_library(grafikr_ffi SHARED IMPORTED GLOBAL)
endif()
if(NOT GRAFIKR_FFI_STATIC AND WIN32 AND GRAFIKR_FFI_IMPORT_LIBRARY)
  set_target_properties(grafikr_ffi PROPERTIES IMPORTED_LOCATION "${GRAFIKR_FFI_LIBRARY}" IMPORTED_IMPLIB "${GRAFIKR_FFI_IMPORT_LIBRARY}")
elseif(NOT GRAFIKR_FFI_STATIC)
  set_target_properties(grafikr_ffi PROPERTIES IMPORTED_LOCATION "${GRAFIKR_FFI_LIBRARY}")
endif()
add_library(grafikr_client STATIC src/client.c src/value.c vendor/cjson/cJSON.c)
target_include_directories(grafikr_client PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
  PRIVATE vendor/cjson)
target_link_libraries(grafikr_client PUBLIC grafikr_ffi PRIVATE Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
  add_library(grafikr_linux_native_deps INTERFACE)
  target_link_libraries(grafikr_linux_native_deps INTERFACE PkgConfig::FONTCONFIG m dl pthread)
  target_link_libraries(grafikr_client PUBLIC grafikr_linux_native_deps)
endif()
if(GRAFIKR_FFI_STATIC)
  target_compile_definitions(grafikr_client PUBLIC GRAFIKR_STATIC)
  if(WIN32)
    target_link_libraries(grafikr_client PUBLIC ws2_32 ntdll dwmapi imm32 uxtheme)
  endif()
endif()
add_executable(grafikr_native_scenario tests/native_scenario.c)
target_include_directories(grafikr_native_scenario PRIVATE include)
target_link_libraries(grafikr_native_scenario PRIVATE grafikr_ffi Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(grafikr_native_scenario PRIVATE grafikr_linux_native_deps)
endif()
if(GRAFIKR_FFI_STATIC)
  target_compile_definitions(grafikr_native_scenario PRIVATE GRAFIKR_STATIC)
  if(WIN32)
    target_link_libraries(grafikr_native_scenario PRIVATE ws2_32 ntdll dwmapi imm32 uxtheme)
  endif()
elseif(WIN32)
  add_custom_command(TARGET grafikr_native_scenario POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GRAFIKR_FFI_LIBRARY}" "$<TARGET_FILE_DIR:grafikr_native_scenario>")
endif()
add_executable(grafikr_collection_provider_scenario tests/collection_provider_scenario.c)
target_link_libraries(grafikr_collection_provider_scenario PRIVATE grafikr_client)
if(NOT GRAFIKR_FFI_STATIC AND WIN32)
  add_custom_command(TARGET grafikr_collection_provider_scenario POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GRAFIKR_FFI_LIBRARY}" "$<TARGET_FILE_DIR:grafikr_collection_provider_scenario>")
endif()
include(GNUInstallDirs)
foreach(example counter_form dashboard multi_window)
  add_executable(grafikr_${example} examples/${example}.c)
  target_link_libraries(grafikr_${example} PRIVATE grafikr_client)
  if(NOT GRAFIKR_FFI_STATIC AND WIN32)
    add_custom_command(TARGET grafikr_${example} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GRAFIKR_FFI_LIBRARY}" "$<TARGET_FILE_DIR:grafikr_${example}>")
  endif()
endforeach()
install(TARGETS grafikr_counter_form grafikr_dashboard grafikr_multi_window
  RUNTIME DESTINATION ${CMAKE_INSTALL_DATADIR}/grafikr/examples)
include(CTest)
if(BUILD_TESTING)
  add_executable(grafikr_abi_test tests/abi.c)
  target_include_directories(grafikr_abi_test PRIVATE include)
  target_link_libraries(grafikr_abi_test PRIVATE grafikr_client)
  if(NOT GRAFIKR_FFI_STATIC)
    add_custom_command(TARGET grafikr_abi_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GRAFIKR_FFI_LIBRARY}" "$<TARGET_FILE_DIR:grafikr_abi_test>")
  endif()
  add_test(NAME grafikr_abi_test COMMAND grafikr_abi_test)
  add_executable(grafikr_client_test tests/client.c)
  target_link_libraries(grafikr_client_test PRIVATE grafikr_client)
  add_test(NAME grafikr_client_test COMMAND grafikr_client_test)
  add_executable(grafikr_value_test tests/value.c)
  target_include_directories(grafikr_value_test PRIVATE vendor/cjson)
  target_link_libraries(grafikr_value_test PRIVATE grafikr_client)
  add_test(NAME grafikr_value_test COMMAND grafikr_value_test)
  add_executable(grafikr_cjson_strict_test tests/cjson_strict.c)
  target_include_directories(grafikr_cjson_strict_test PRIVATE vendor/cjson)
  target_link_libraries(grafikr_cjson_strict_test PRIVATE grafikr_client)
  add_test(NAME grafikr_cjson_strict_test COMMAND grafikr_cjson_strict_test)
  add_library(grafikr_test_mock_ffi STATIC tests/mock_ffi.c)
  target_include_directories(grafikr_test_mock_ffi PRIVATE include tests vendor/cjson)
  target_compile_definitions(grafikr_test_mock_ffi PRIVATE GRAFIKR_STATIC)
  target_link_libraries(grafikr_test_mock_ffi PRIVATE Threads::Threads)
  add_library(grafikr_client_mock STATIC src/client.c src/value.c vendor/cjson/cJSON.c)
  target_include_directories(grafikr_client_mock PUBLIC include PRIVATE vendor/cjson)
  target_compile_definitions(grafikr_client_mock PRIVATE GRAFIKR_STATIC)
  target_link_libraries(grafikr_client_mock PRIVATE grafikr_test_mock_ffi Threads::Threads)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(grafikr_client_mock PRIVATE grafikr_linux_native_deps)
  endif()
  add_executable(grafikr_transport_contract_test tests/transport_contract.c)
  target_compile_definitions(grafikr_transport_contract_test PRIVATE GRAFIKR_STATIC)
  target_link_libraries(grafikr_transport_contract_test PRIVATE grafikr_client_mock grafikr_test_mock_ffi Threads::Threads)
  add_test(NAME grafikr_transport_contract_test COMMAND grafikr_transport_contract_test)
endif()
install(TARGETS grafikr_client ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/grafikr_client.h ../c/include/grafikr.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES vendor/cjson/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/grafikr/cjson)
